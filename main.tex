\section{Prima lezione}

\begin{definizione}[Media campionaria]
Dati $x_1,\ldots,x_n$, la media campionaria è
\[
\overline{x}=\frac{1}{n}\sum_{i=1}^{n}x_i.
\]
\end{definizione}

\begin{teorema}
Se $X_1,\ldots,X_n$ hanno lo stesso valore atteso finito $\mu$,
allora $\mathbb{E}[\overline{X}]=\mu$.
\end{teorema}

\begin{esempio}
Per i dati $2,4,6$ la media è $\overline{x}=4$.
\end{esempio}

\begin{esercizio}
Calcolare la media di $1,3,5,7$.
\end{esercizio}
