\section{Esempi degli ambienti}

\begin{definizione}
La media aritmetica di $x_1,\ldots,x_n$ è
\[\overline{x}=\frac{1}{n}\sum_{i=1}^n x_i.\]
\end{definizione}

\begin{teorema}
Se $X_1,\ldots,X_n$ hanno lo stesso valore atteso $\mu$,
allora $\mathbb{E}[\overline{X}]=\mu$.
\end{teorema}

\begin{esempio}
La media dei numeri $2,4,6$ è $4$.
\end{esempio}

\begin{esercizio}
Calcolare la media dei numeri $1,3,5,7$.
\end{esercizio}

\begin{approfondimento}
Qui si può aggiungere una spiegazione personale.
\end{approfondimento}
